def patch_func(owner, name, old, new, module):
    fn = getattr(owner, name)
    src = textwrap.dedent(inspect.getsource(fn))
    assert old in src, (name, old)
    src = src.replace(old, new)
    ns = {}
    exec(compile(src, f"<mut {name}>", "exec"), module.__dict__, ns)
    orig = fn
    setattr(owner, name, ns[name] if not isinstance(inspect.getattr_static(owner, name, None), staticmethod) else staticmethod(ns[name]))
    return orig

def run(ids):
    loader = unittest.TestLoader()
    suite = unittest.TestSuite()
    for i in ids:
        suite.addTests(loader.loadTestsFromName(i))
    res = unittest.TextTestRunner(verbosity=0, stream=open("/dev/null", "w")).run(suite)
    return res.testsRun, len(res.failures), len(res.errors)

# A-03 greedy split mutation: replace partition_valid everywhere
from decimal import Decimal
import math
from fractions import Fraction
orig_pv = grid.partition_valid
def greedy(qty, price, rules):
    step = rules.size_step
    n = math.floor(Fraction(qty) / Fraction(step))
    m = int(Fraction(grid.min_valid_order_qty(rules, price)) / Fraction(step))
    if n < m: return []
    if rules.max_base is None: chunks = [n]
    else:
        big = math.floor(Fraction(rules.max_base) / Fraction(step))
        chunks = []
        left = n
        while left >= m:
            c = min(left, big); chunks.append(c); left -= c
            if c < m: chunks.pop(); break
    out = [Decimal(c) * step for c in chunks]
    return [qty if c == qty else c for c in out]
for mod in (grid, cells, dust):
    if hasattr(mod, "partition_valid"): setattr(mod, "partition_valid", greedy)
r = run([TPKG + ".test_cells", TPKG + ".test_dust"])
for mod in (grid, cells, dust):
    if hasattr(mod, "partition_valid"): setattr(mod, "partition_valid", orig_pv)
print(f"{'A-03 greedy split':40s} run={r[0]} fail={r[1]} err={r[2]} -> {'KILLED' if r[1] or r[2] else 'SURVIVED'}")

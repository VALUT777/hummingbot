def _rewrite(monkeypatch, name, pairs):
    src = textwrap.dedent(inspect.getsource(getattr(E, name)))
    for old, new in pairs:
        assert old in src, (name, old)
        src = src.replace(old, new)
    ns = {}
    exec(compile(src, f"<mut {name}>", "exec"), eng.__dict__, ns)
    monkeypatch.setattr(E, name, ns[name])


def _expect_fail(fn, *args):
    try:
        fn(*args)
    except AssertionError as exc:
        print(f"KILLED {fn.__name__}: {str(exc)[:160]!r}")
        return
    raise AssertionError(f"SURVIVED: {fn.__name__} passed under the mutation")

def test_foreign_market_trade_signal(tmp_path):
    h = Harness(tmp_path)
    try:
        h.bootstrap(D("0"))
        h.tick(3)
        # An OrderFilledEvent of another trading pair on the same connector (executor forwards it unfiltered).
        h.engine.wake({"type": "trade", "status": "filled", "client_order_id": "HBOT-other-pair-1",
                       "trade_id": "987654321"})
        h.tick(120)
        print("history_complete", h.engine.history_complete, "lag_s", h.engine.history_lag_s(),
              "pending", h.engine.ws_pending)
        print("audit unsettled:", h.engine._audit_evidence_unsettled(h.clock()))
        print("snapshot lag:", h.engine.last_snapshot["summary"]["history"].get("lag_s"))
    finally:
        h.close()

# A-03 exhaustive maximality: partition_valid vs brute-force optimum (step 1)
bad = 0
for m in range(1, 8):
    for big in range(m, 15):
        r = rules(step="1", min_base=str(m), min_notional="0", max_base=str(big))
        for n in range(0, 60):
            parts = grid.partition_valid(D(n), D("5"), r)
            assert all(m <= p <= big for p in parts), (m, big, n, parts)
            got = sum(parts, D(0))
            # best achievable: max total <= n expressible as sum of k values in [m,big]
            best = 0
            for k in range(0, n // m + 1):
                if k * m <= n:
                    best = max(best, min(n, k * big))
            if got != best:
                bad += 1
                print("A-03 suboptimal", m, big, n, parts, best)
print("A-03 partition bad cases:", bad)

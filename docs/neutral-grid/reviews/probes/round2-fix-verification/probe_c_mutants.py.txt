def run(names):
    loader = unittest.TestLoader()
    suite = unittest.TestSuite()
    for n in names:
        suite.addTests(loader.loadTestsFromName(n))
    res = unittest.TextTestRunner(verbosity=0, stream=open("/dev/null", "w")).run(suite)
    return len(res.failures) + len(res.errors), res.testsRun


# ---- C-06 mutant: retry with a new CID after a transport exception
orig_submit = LighterPerpetualDerivative.submit_with_client_id


async def mutant_submit(self, client_order_id, *a, **kw):
    result = await orig_submit(self, client_order_id, *a, **kw)
    if result.outcome == mod.LighterTransportOutcome.UNKNOWN and result.detail.startswith("transport_exception"):
        try:
            new_cid = int(self._new_client_order_id())
            await self._signer_client.create_order(client_order_index=new_cid)
        except Exception:
            pass
    return result

LighterPerpetualDerivative.submit_with_client_id = mutant_submit
print("C-06 mutant (retry new CID) failures/run:",
      run([f"{conn_tests}.LighterHistoryPaginationTest.test_ac56_timeout_not_found_and_errors_are_unknown_and_never_retry_with_new_cid"]))

# ---- C-01/C-05 mutant: _result ignores conflicted keys
orig_result = hist.HistoryScanner._result


def mutant_result(self, complete, reason, pages, weight, conflicts=None, high_water=None, emit_rows=False):
    saved = {}
    for p in (self._progress or {}).values():
        saved[p.stream] = p.conflicted_keys
        p.conflicted_keys = {}
    try:
        return orig_result(self, complete, reason, pages, weight, conflicts=conflicts, high_water=high_water,
                           emit_rows=emit_rows)
    finally:
        for p in (self._progress or {}).values():
            p.conflicted_keys = saved[p.stream]

hist.HistoryScanner._result = mutant_result
print("C-01/C-05 mutant (no withholding) failures/run:",
      run([f"{scan_tests}.ConflictWithholdingTest",
           f"{scan_tests}.HistoryScannerTest.test_ac12_bad_pagination_makes_history_incomplete",
           f"{scan_tests}.HistoryScannerTest.test_ac40_trades_exceeding_terminal_order_cumulative_is_conflict"]))
hist.HistoryScanner._result = orig_result

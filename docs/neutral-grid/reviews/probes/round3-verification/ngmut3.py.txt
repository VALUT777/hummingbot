def _apply(name, repls):
    fn = getattr(E.NeutralGridEngine, name)
    src = textwrap.dedent(inspect.getsource(fn))
    for old, new in repls:
        if old not in src:
            raise RuntimeError(f"mutation anchor not found in {name}: {old!r}")
        src = src.replace(old, new, 1)
    ns = {}
    exec(compile(src, f"<mut:{name}>", "exec"), E.__dict__, ns)
    setattr(E.NeutralGridEngine, name, ns[name])


for mut in filter(None, os.environ.get("MUT", "").split(",")):
    if mut == "d116_both":
        _apply("_tick_body", MUTATIONS["d116_nopersist"][1])
        _apply("_recover_after_refusal", [("self._reload()\n", "pass\n")])
    else:
        name, repls = MUTATIONS[mut]
        _apply(name, repls)
    print(f"[ngmut3] applied {mut}")

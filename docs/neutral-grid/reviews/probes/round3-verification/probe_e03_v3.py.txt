for stale_after in (15.0, 60.0, 5.0):
    tmp = Path(tempfile.mkdtemp(dir="/private/tmp/claude-501/-Users-mikhail/ed31a3df-55bd-475b-8111-912d06f80d1d/scratchpad/verify3"))
    h = Harness(tmp)
    data_dir = h.db_path.parent
    try:
        h.bootstrap(D("0")); h.tick(25)   # healthy: heartbeat sidecar written
        side = data_dir / (h.db_path.name + ".health.json")

        snap = h.engine.latest_committed_snapshot()
